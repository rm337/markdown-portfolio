,text='';
   if(state.resume.text)backup('Automatic backup before replace');
   if(ext==='pdf')text=await extractPDF(file);
   else if(ext==='docx')text=await extractDOCX(file);
   else if(['txt','md'].includes(ext)||file.type.startsWith('text/'))text=await file.text();
   else if(ext==='rtf')text=(await file.text()).replace(/\\par[d]?/g,'\n').replace(/\\'[0-9a-fA-F]{2}/g,' ').replace(/\\[a-zA-Z]+-?\d* ?/g,'').replace(/[{}]/g,' ');
   else throw new Error('Unsupported file type.');
   text=text.trim();if(!text)throw new Error('No readable résumé text was extracted.');
   state.resume={name:file.name,type:ext,text,loadedAt:new Date().toLocaleString(),analysis:analyze(text)};save();render();
 }catch(e){alert('Resume load failed: '+e.message)}
}

function fsttStageIndex(a){
 const map={'Interested':0,'Preparing':0,'Applied':1,'Follow-Up':4,'Interview':5,'Offer':6,'Closed':7};
 if(Number.isInteger(a.journeyStage)) return Math.max(0,Math.min(7,a.journeyStage));
 return map[a.status] ?? 0;
}
function fsttConfidence(i,current){
 if(i<current) return 'CONFIRMED';
 if(i===current) return current<=1?'CONFIRMED':'DETECTED';
 if(i===current+1) return 'INFERRED';
 return 'UNKNOWN';
}
