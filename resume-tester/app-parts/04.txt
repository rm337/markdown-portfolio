}$/.test(l));
 const metricRx=/(?:\b\d+(?:\.\d+)?\s*%|\$\s?\d[\d,.]*(?:\s?[KMB])?|\b\d+(?:\.\d+)?\s?(?:x|times|users|clients|customers|projects|campaigns|teams|people|employees|stores|locations|markets|countries|years|months|days|hours|units|assets|pieces|deliverables|accounts|leads|views|impressions|followers|subscribers|sales|revenue)\b)/i;
 const leadershipRx=/\b(led|lead|directed|managed|oversaw|mentored|trained|supervised|headed|owned|guided|coached|built a team|cross-functional|stakeholders?)\b/i;
 const strategyRx=/\b(strategy|strategic|positioning|roadmap|vision|architecture|framework|go-to-market|campaign|brand system|decision system|priorit|audience|market|research)\b/i;
 const creationRx=/\b(created|built|designed|developed|launched|produced|wrote|crafted|concepted|prototyped|implemented|established)\b/i;
 const impactRx=/\b(increased|grew|reduced|cut|saved|improved|boosted|generated|delivered|drove|transformed|optimized|accelerated|expanded|raised|lowered|won|sold)\b/i;
 const vagueRx=/\b(innovative|dynamic|results[- ]driven|passionate|hard[- ]working|team player|go[- ]getter|excellent|strong communicator|detail[- ]oriented|creative thinker)\b/i;
 const passiveRx=/\b(responsible for|helped with|worked on|duties included|assisted with|participated in|involved in)\b/i;

 const evidence=contentLines.map(line=>{
   let score=0, tags=[];
   if(metricRx.test(line)){score+=3;tags.push('MEASURE')}
   if(impactRx.test(line)){score+=2;tags.push('IMPACT')}
   if(leadershipRx.test(line)){score+=2;tags.push('LEADERSHIP')}
   if(strategyRx.test(line)){score+=1;tags.push('STRATEGY')}
   if(creationRx.test(line)){score+=1;tags.push('CREATION')}
   if(passiveRx.test(line)){score-=2;tags.push('PASSIVE')}
   if(vagueRx.test(line)){score-=1;tags.push('VAGUE')}
   return {line,score,tags};
 });

 const leadership=evidence.filter(x=>x.tags.includes('LEADERSHIP'));
 const strategy=evidence.filter(x=>x.tags.includes('STRATEGY'));
 const impact=evidence.filter(x=>x.tags.includes('IMPACT'));
 const measured=evidence.filter(x=>x.tags.includes('MEASURE'));
 const passive=evidence.filter(x=>x.tags.includes('PASSIVE'));
 const vague=evidence.filter(x=>x.tags.includes('VAGUE'));
 const strong=evidence.filter(x=>x.score>=4).sort((a,b)=>b.score-a.score).slice(0,6);
 const buried=evidence.filter(x=>x.score>=3 && x.line.length>165).slice(0,6);
 const unproved=evidence.filter(x=>x.tags.some(t=>['LEADERSHIP','STRATEGY','CREATION'].includes(t)) && !x.tags.includes('MEASURE') && !x.tags.includes('IMPACT')).slice(0,8);

 // Similarity check for repeated ideas without requiring identical wording.
 const tokenize=s=>new Set((s.toLowerCase().match(/[a-z][a-z0-9-]{3,}/g)||[]).filter(w=>!['with','that','from','this','into','through','using','work','role','team','creative','design'].includes(w)));
 const sim=(a,b)=>{let A=tokenize(a),B=tokenize(b); if(!A.size||!B.size)return 0; let inter=[...A].filter(x=>B.has(x)).length; return inter/Math.min(A.size,B.size)};
 let duplicates=[];
 for(let i=0;i<contentLines.length;i++){
   for(let j=i+1;j<contentLines.length;j++){
     if(Math.abs(i-j)<2)continue;
     const v=sim(contentLines[i],contentLines[j]);
     if(v>=0.58)duplicates.push({a:contentLines[i],b:contentLines[j],similarity:v});
   }
 }
 duplicates.sort((a,b)=>b.similarity-a.similarity);
 duplicates=duplicates.slice(0,5);

 // Seniority signal based on actual language, not title.
 let seniorityScore=leadership.length*2+strategy.length+impact.length+(normalized.match(/\b(executive|director|senior|head of|vice president|vp|chief)\b/gi)||[]).length*2;
 let seniority=seniorityScore>=18?'EXECUTIVE / SENIOR LEADERSHIP SIGNAL':seniorityScore>=10?'SENIOR / LEADERSHIP SIGNAL':seniorityScore>=5?'MID-SENIOR SIGNAL':'LIMITED SENIORITY SIGNAL';

 // Compare against most recent saved job when available.
 let jobMatch=null;
 const job=state.jobs&&state.jobs[0];
 if(job?.description){
   const stop=new Set(['about','after','again','against','also','and','are','because','been','being','between','both','but','can','company','could','does','each','from','have','into','job','more','most','other','our','role','that','their','them','then','there','these','they','this','through','under','very','what','when','where','which','while','will','with','would','your']);
   const words=t=>(t.toLowerCase().match(/[a-z][a-z0-9+.#/-]{2,}/g)||[]).filter(w=>!stop.has(w));
   const freq=arr=>arr.reduce((m,w)=>(m[w]=(m[w]||0)+1,m),{});
   const jf=freq(words(job.description)), rf=freq(words(normalized));
   const important=Object.entries(jf).filter(([w,c])=>c>=2 || /\b(strategy|brand|creative|content|product|design|leadership|campaign|social|marketing|director|manager|experience|research|production|growth|team)\b/.test(w))
     .sort((a,b)=>b[1]-a[1]).slice(0,35).map(([w])=>w);
   const matched=important.filter(w=>rf[w]);
   const missing=important.filter(w=>!rf[w]);
   const score=important.length?Math.round(matched.length/important.length*100):0;
   jobMatch={title:job.title||'Saved job',company:job.company||'',score,matched:matched.slice(0,14),missing:missing.slice(0,14)};
 }

 const evidenceCoverage=Math.min(100,Math.round((measured.length*4+impact.length*3+leadership.length*2+strategy.length)/(Math.max(1,contentLines.length)*3)*100));
 return {leadership,strategy,impact,measured,passive,vague,strong,buried,unproved,duplicates,seniority,seniorityScore,jobMatch,evidenceCoverage};
}
