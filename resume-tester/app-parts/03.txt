function professionalResumeExplanation(a){
 if(!a||!a.intelligence)return '';
 const i=a.intelligence;
 const strengths=[];
 if(i.seniority.includes('EXECUTIVE')) strengths.push('senior creative leadership');
 else if(i.seniority.includes('SENIOR')) strengths.push('senior-level responsibility');
 if(i.strategy.length>=5) strengths.push('strategic thinking');
 if(i.leadership.length>=2) strengths.push('leadership and ownership');
 if(i.strong.length) strengths.push('specific high-value experience');
 if(!strengths.length) strengths.push('a broad range of professional capabilities');

 const proof=[];
 if(i.measured.length) proof.push('measurable results');
 if(i.impact.length) proof.push('outcome language');
 if(i.leadership.length) proof.push('leadership evidence');
 if(i.strategy.length) proof.push('strategic evidence');

 let opening=`Your résumé presents you as a professional with ${strengths.join(', ').replace(/, ([^,]*)$/, ' and $1')}. It communicates the kind of work you perform and the level of responsibility you have held.`;
 let improvement='';
 if(!i.measured.length && !i.impact.length){
   improvement=`The main opportunity is to show more clearly what changed because of your work. The résumé currently explains your capabilities more strongly than it documents their outcomes. Where accurate, examples such as project scale, audience growth, sales or revenue influence, engagement, time saved, turnaround improvements, team size, volume of work, or other concrete results would make the existing experience more persuasive.`;
 }else if(!i.measured.length){
   improvement=`The résumé contains outcome-oriented language, but relatively little quantified proof. Where accurate, adding scale, percentages, dollar impact, audience size, team size, volume, time saved, or similar measures would make those outcomes easier to evaluate.`;
 }else{
   improvement=`The résumé contains measurable evidence as well as broader leadership and strategic signals. The next opportunity is to make sure the strongest results are easy to find and directly connected to the work that produced them.`;
 }
 let clarification=`This does not mean the résumé lacks experience or value. It means the current document ${(!i.measured.length&&!i.impact.length)?'describes your capabilities more clearly than it demonstrates their results':'has evidence that can be presented more deliberately and prominently'}.`;
 let job='';
 if(i.jobMatch){
   job=`For the saved target role, the résumé currently reflects ${i.jobMatch.matched.length?'several':'few'} of the recurring concepts found in the posting. The comparison should be used to locate relevant experience that may be buried or described differently, not to add claims that are not true.`;
 }
 return {opening,improvement,clarification,job};
}
