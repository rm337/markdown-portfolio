function detectJobFromText(text,url=''){
 const lines=String(text||'').split(/\n/).map(x=>x.trim()).filter(Boolean);
 const salary=(String(text).match(/\$\s?\d[\d,]*(?:\s*[-–]\s*\$?\s*\d[\d,]*)?(?:\s*(?:per year|annually|\/year|yearly))?/i)||[])[0]||'';
 const location=(String(text).match(/\b(?:Remote|Hybrid|On-site|Onsite)\b/i)||[])[0]||'';
 let title=lines.find(l=>/\b(?:director|manager|designer|writer|strategist|lead|engineer|specialist|coordinator|executive|president|officer)\b/i.test(l)&&l.length<100)||'';
 let company='';
 const companyLine=lines.find(l=>/^(?:company|organization|employer)\s*:/i.test(l));
 if(companyLine) company=companyLine.replace(/^[^:]+:\s*/, '').trim();
 return {description:String(text||''),url:String(url||''),title,company,location,salary};
}
async function extractDOCX(file){
 if(!window.mammoth||typeof window.mammoth.extractRawText!=='function'){
   await new Promise((resolve,reject)=>{
     const s=document.createElement('script');
     s.src='https://cdn.jsdelivr.net/npm/mammoth@1.8.0/mammoth.browser.min.js';
     s.async=true;
     s.onload=resolve;
     s.onerror=()=>reject(new Error('DOCX parser failed to load'));
     document.head.appendChild(s);
   });
 }
 if(!window.mammoth||typeof window.mammoth.extractRawText!=='function') throw new Error('DOCX parser failed to initialize');
 const r=await window.mammoth.extractRawText({arrayBuffer:await file.arrayBuffer()});
 return (r&&r.value)||'';
}
